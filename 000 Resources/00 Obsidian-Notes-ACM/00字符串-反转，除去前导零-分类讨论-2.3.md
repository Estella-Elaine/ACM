#include<bits/stdc++.h>  
#define all(a) (a).begin(), (a).end()  
using namespace std;  
using ll = long long; 
//优化代码：  
//void reverse(string x){

void work(){  
    ll l, r;  
    cin >> l >> r;  
    auto x = to_string(r);  
        //如果l == r，直接反转输出  
    if(l == r){  
        reverse(all(x)); //更优写法：ranges::reverse(t);  
        cout << stoll(x) <<'\n'; //stoll(String To Long Long)：这是一个标准库函数，将字符串转换为 long long 类型的数值。  
        //处理前导零：这是该代码的关键点。stoll 会自动忽略字符串开头的零。  
        return;  
    }  
        //如果r是10的幂次且r>l，则直接输出r-1(若干个9)  
    if(x[0] == '1' && count(all(x), '0') == x.size()-1){  
        cout << r-1 << endl;  
        return;  
    }  
        //否则，构造  
    ll ans = stoll("1" + string(x.size()-1, '0')); //构造一个基准：与 r 同位数的最小整数（比r小的最大的10的幂次）  
    ans = max(ans, l); //确保了我们构造的数始终在范围 [l,r] 内  
    auto s = string(18, '0') + to_string(ans);  
    ll n = s.size();  
    for(ll i=1, j=n-1; i<=1e16; i*=10, j--){  
        for(int k = 1; k <= '9' - s[j]; k++){ //可读性更高（maybe  
            if(ans + i <= r) ans += i;  
            else break;  
        }  
    }  
    s = to_string(ans);  
    reverse(all(s));  
    cout << s << endl;  
}  
signed main(){  
    ios::sync_with_stdio(0);  
    cin.tie(0);  
    int T=1;  
    cin >> T;  
    while(T--) work();
}
