[PTA团体程序设计天梯赛-练习集 L3-003 社交集群（并查集） - 代码先锋网](https://www.codeleading.com/article/51574038484/)
```cpp
#include<bits/stdc++.h>
#define all(a) (a).begin(), (a).end()
using namespace std;
using ll = long long;
const int M = 1005;
int par[M], hob[M], cnt[M]; 
// 并查集数组，记录某个爱好第一次分配给了哪个人，记录每个集群（以根节点为代表）的人数

// 查找根节点（带路径压缩）
int find(int x){
    if(x == par[x]) return x;
    return par[x] = find(par[x]);
}

// 合并两个集合
void unity(int x, int y){
    int rx = find(x);
    int ry = find(y);
    if(rx != ry) par[rx] = ry;
}

void work(){
    int n;
    cin >> n;
    for(int i=1; i<=n; i++) par[i] = i; // 初始化并查集
    for(int i=1; i<=n; i++){
        int k;
        scanf("%d: ", &k);
        for(int j=0; j<k; j++){
            int x;
            cin >> x;
            // 如果这个爱好之前没有出现过，记录下第一次有这个爱好的人是谁
            if(hob[x] == 0) hob[x] = i;
            else unity(i, hob[x]);
        }
    }
	
	// 统计每个集群的人数
    for(int i=1; i<=n; i++){
        int r = find(i);
        ++cnt[r];
    }
	
	// 提取出有效的集群人数
    vector<int> ans;
    for(int i=1; i<=n; i++){
        if(cnt[i]) ans.push_back(cnt[i]);
    }

    sort(all(ans), greater<int>());

    int sz = ans.size();
    cout << sz << '\n';

    for(int i=0; i<sz; i++){
        if(i) cout << " ";
        cout << ans[i];
    }
}

signed main(){
    // ios::sync_with_stdio(0);
    // cin.tie(0);
    int T=1;
//  cin >>T;
    while(T--) work();
}
```

[P1195 口袋的天空 - 洛谷](https://www.luogu.com.cn/problem/P1195)
```cpp
#include<bits/stdc++.h>  
#define all(a) (a).begin(), (a).end()  
using namespace std;  
using ll = long long;  
const int M = 1005;  
int fa[M];  
  
struct o{  
    int x, y, l;  
};  
  
int find(int x){  
    if(fa[x] == x) return x;  
    return fa[x] = find(fa[x]);  
}  
  
void unity(int x, int y){  
    int rx = find(x);  
    int ry = find(y);  
    if(rx != ry) fa[rx] = ry;  
}  
  
void work(){  
    int n, m, k;  
    cin >> n >> m >> k;  
    for(int i=1; i<=n; i++) fa[i] = i; // 0-base  
    vector<o> a(m);  
    for(int i=0; i<m; i++) cin >> a[i].x >> a[i].y >> a[i].l;  
    
    if(n < k){  
        cout << "No Answer\n";  
        return;  
    }  
    if(n == k){  
        cout << "0\n";  
        return;  
    }  
    
    sort(all(a), [](const o& a, const o& b){  
        return a.l < b.l;  
    });  
    ll ans = 0;  
    int i=0, cnt = n;  
    for(i=0; i<m; i++){  
        if(find(a[i].x) != find(a[i].y)){    // 排除重边  
            unity(a[i].x, a[i].y);  
            ans += a[i].l;  
            --cnt;  
            if(cnt == k) break;  
        }  
    }  
    
    if(i == m) cout << "No Answer\n";  
    else cout << ans << '\n';  
}  
  
signed main(){  
    ios::sync_with_stdio(0);  
    cin.tie(0);  
    int T=1;  
//    cin >>T;  
    while(T--) work();  
}  
  
/*  
这段代码的目的是使用 Kruskal算法 求出将图连通至恰好剩下 k 个连通块时的最小边权和  
Kruskal 算法的核心是每次要选择属于不同连通块的两个点连边
```

[P1141 01迷宫 - 洛谷](https://www.luogu.com.cn/problem/P1141)
```cpp
#include<bits/stdc++.h>  
#define all(a) (a).begin(), (a).end()  
using namespace std;  
using ll = long long;  
const int M = 1000005;  
int fa[M];  
int n;  
  
int gid(int x, int y){  
    return y+(x-1)*n;  
}  
  
int find(int x){  
    if(fa[x] == x) return x;  
    return fa[x] = find(fa[x]);  
}  
  
void unity(int x, int y){  
    int rx = find(x);  
    int ry = find(y);  
    if(rx != ry) fa[rx] = ry;  
}  
  
void work(){  
    int m;  
    cin >> n >> m;  
    vector<string> g(n+1);  
    for(int i=1; i<=n; i++){  
        cin >> g[i];  
        g[i] = " " + g[i];  
        for(int j=1; j<=n; j++){  
            fa[gid(i, j)] = gid(i, j);  
            if(i > 1 && g[i-1][j] != g[i][j]) unity(gid(i-1, j), gid(i, j));  
            if(j > 1 && g[i][j-1] != g[i][j]) unity(gid(i, j-1), gid(i, j));  
        }  
    }  
        vector<int> cnt(n*n+1, 0);  
    for(int i=1; i<=n*n; i++){  
        ++cnt[find(i)];  
    }  
        while(m--){  
        int x, y;  
        cin >> x >> y;  
        cout << cnt[find(gid(x, y))] << '\n';  
    }  
}  
  
signed main(){  
    ios::sync_with_stdio(0);  
    cin.tie(0);  
        int T=1;  
//    cin >>T;  
    while(T--) work();  
}
```
