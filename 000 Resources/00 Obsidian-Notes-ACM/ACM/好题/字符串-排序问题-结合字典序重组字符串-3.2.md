**题面：** 设有n个正整数，将他们连接成一排，组成一个最大的多位整数。
如:n=3时，3个整数13,312,343连成的最大整数为34331213。
如:n=4时，4个整数7,13,4,246连接成的最大整数为7424613。  
**输入格式:**  
有多组测试样例，每组测试样例包含两行，第一行为一个整数N（N<=100），第二行包含N个数(每个数不超过1000，空格分开)。  
**输出格式:**
每组数据输出一个表示最大的整数。

```cpp
#include<bits/stdc++.h>  
using ll = long long;  
#define all(a) (a).begin(), a.end()  
#define pii pair<int, int>  
using namespace std;  
  
void work(){  
    int n;  
    int flag = 0;  
    while(cin >> n){  
        if(flag) cout << " " << endl;  
        vector<string> a(n);  
        for(int i=0; i<n; i++) cin >> a[i];  
        
        //拼接字符串使得最后的字符串最大时，
        //排序应该保证s1+s2 > s2+s1,而不是仅仅按照字典序 s1>s2!!!
        sort(all(a), [](const string& s1, const string& s2){  
            return s1+s2 < s2+s1;  
        });  
        
        string ans = "";  
        for(int i=n-1; i>=0; i--){  
            ans += a[i];  
        }  
        cout << ans;  
        flag = 1;  
    }  
}  
  
signed main(){  
    ios::sync_with_stdio(0);  
    cin.tie(0);  
    cout.tie(0);  
    int T=1;  
//    cin >> T;  
    while(T--) work();  
}

/*
2 
12 123 
4 
7 13 4 246

12312 
7424613
*/
```

##### **注意！！！
##### 拼接字符串使得最后的字符串最大时，排序应该保证s1+s2 > s2+s1,
##### 而不是仅仅按照字典序 s1>s2!!!**

###### 另，注意，如果这么写：

```cpp
for(int i=n-1; i>=0; i--){

}
```
