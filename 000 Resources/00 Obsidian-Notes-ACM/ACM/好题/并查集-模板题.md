#include<bits/stdc++.h>
using namespace std;
const int Max = 1e6+5;
int z[Max], x[Max], y[Max];
int fa[Max];

int Find(int x){
	if(fa[x] == x) return x;
	return fa[x] = Find(par[x]);
	//虽然第一次找老大的路途可能很漫长，但只要走过一次，这条路就被永久“压平”了。
}

void Union(int x, int y){
	int rootx = Find(x);
	int rooty = Find(y);
	if(rootx != rooty) fa[rootx]=rooty;

}

void work(){
	int n, m;
	cin >> n >> m;
	
	for(int i=1; i<=m; i++) fa[i] = i;
	
	for(int i=1; i<=m; i++){
		cin >> z[i] >> x[i] >> y[i];
		if(z[i] == 1){
			Union(x[i], y[i]);
		}
		else if(z[i] == 2){
			if(Find(x[i]) == Find(y[i])){
				cout << "Y\n";
			}
			else{
				cout << "N\n";
			}
		}
	}
}

signed main(){
	ios::sync_with_stdio(0);
	cin.tie(0);
	
	int T=1;
//	cin >> T;
	while(T--){
		work();
	}
}
