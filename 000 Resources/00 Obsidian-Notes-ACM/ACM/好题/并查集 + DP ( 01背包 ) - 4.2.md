#include<bits/stdc++.h>  
#define all(a) (a).begin(), (a).end()  
using namespace std;  
using ll = long long;  
const int M = 100005;  
int fa[M];  
  
struct o{  
    ll c, d, s, v;  
};  
  
int find(int x){  
    if(fa[x] == x) return x;  
    return fa[x] = find(fa[x]);  
}  
  
void unity(int x, int y){  
    int rx = find(x);  
    int ry = find(y);  
    if(rx != ry) fa[rx] = ry;  
}  
  
void work(){  
    int n, m, w;  
    cin >> n >> m >> w;  
    vector<o> a(n+1);  
    for(int i=1; i<=n; i++) cin >> a[i].c >> a[i].d;
    for(int i=1; i<=n; i++) fa[i] = i;  
    for(int i=0; i<m; i++){  
        int u, v;  
        cin >> u >> v;  
        unity(u, v);  
    }  
    
    //因为那些没有被直接 find 访问到的底层子节点，  
    //它们的 fa 数组存的依然是旧的上级，而不是最新的终极老大。  
    
    //所以在所有合并操作完成后，如果你要统计每个帮派的总人数、总资产等信息，  
    //绝对不能用 fa[i] 来分组，必须对所有节点统一再调用一次 find(i)，  
    //强制它们去确认最新的终极老大到底是谁。  
    vector<o> ans(n+1);  
    for(int i=1; i<=n; i++){  
        int r = find(i); // 一定不能直接用 fa[i] ！！！  
        ans[r].s += a[i].c;  
        ans[r].v += a[i].d;  
    }  
    
    vector<ll> dp(w+1, 0);  
    for(int i=1; i<=n; i++){  // 01背包
        if(ans[i].s){  
            for(int j=w; j >= ans[i].s; j--){  
                dp[j] = max(dp[j-ans[i].s] + ans[i].v, dp[j]);  
            }  
        }  
    }  
    cout << dp[w] << '\n';  
}  
  
signed main(){  
    ios::sync_with_stdio(0);  
    cin.tie(0);  
    int T=1;  
//    cin >>T;  
    while(T--) work();  
}
