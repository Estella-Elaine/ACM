#include<bits/stdc++.h>  
#define all(a) (a).begin(), (a).end()

using ll = long long; void work(){  
    int n;  
    cin >> n;  
    int m = 1 << n;  
    for(int i=0; i<m; ++i){  
        //二进制反射格雷码（Gray code）：g(i) = i ^ (i >> 1)  
        int ans = i ^ (i >> 1);
        cout << ans << " ";  
    }  
}  
signed main(){  
    ios::sync_with_stdio(0);  
    cin.tie(0);  
    int t=1;  
//  cin >> t;  
    while(t--) work();
}
