```cpp
#include<bits/stdc++.h>  
using namespace std;  
using ll = long long;
ll max_happy(ll n, ll a, ll b, ll k){ //枚举"qcjjkkt"的数量k  
    if(k < 0 || k*7 > n) return 0;  
    ll sum7 = k*a;  
    ll rem = n - k*7;  
	sum7 += min(rem, k) * b;  
    rem -= min(rem, k);  
    sum7 += rem/2*b;  
	return sum7;  
}  
  
void work(){  
    ll n, a, b;  
    cin >> n >> a >> b;  
    ll ans = 0;  
    ans = max(ans, max_happy(n, a, b, 0)); //全"td"  
    ans = max(ans, max_happy(n, a, b, n/7)); //全"qcjjkkt"  
    ans = max(ans, max_happy(n, a, b, n/8)); //全2+7结合:"qcjjkktd" 向下取整
    ans = max(ans, max_happy(n, a, b, n/8 + 1)); //全2+7结合:"qcjjkktd" 向上取整
    cout << ans << '\n';  
}  
signed main(){  
    ios::sync_with_stdio(0);  
    cin.tie(0);  
	int t=1;  
    cin >> t;  
    while(t--) work();
}  
  
/*  
3  
10 9 1  
1000000000 1000000000 1000000000  
1 1 4  
  
11  
500000000000000000  
0  
*/
```

#### 注意！不要忘记考虑峰值为小数时，向上取整的情况！

##### 思路重现：
这是一个经典的优化问题（类似背包问题的变种），可以通过分析数学性质用 O(1)O(1) 的贪心策略解决。
##### 解题关键：可以发现，只需要检查4个关键的k值即可找到最大值（尤其注意第四个！）
因为总贡献关于 k 的函数是分段线性且具有**单峰性**（或者说**最大值一定在边界取到**），因此不需要遍历所有 k，我们可以定性分析：
**·如果a>>b 则 k 要尽可能大 （k = n/7)**
**·如果b>>a 则 k 要尽可能小   (k = 0)**
**·中间可能存在一个平衡点**，即“尽可能多选 k，但前提是每一个 k 都能接上一个优惠的 td 变成qcjjkktd”。**这个临界点发生在剩余空间刚好够填满接口时，
即 n−7k≈k⇒8k≈n⇒k≈n/8n−7k≈k⇒8k≈n⇒k≈n/8。**

至于为什么要判断第四个(k = n/8 + 1 向上取整) ，因为 n/8 可能是小数，需要同时判断 **向下取整** 和 **向上取整** 这两个整数来寻找**最优**
